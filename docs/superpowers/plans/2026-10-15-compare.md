# Chuyên mục Từ dễ nhầm — Kế hoạch triển khai

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Thêm chuyên mục `⚖️ Từ dễ nhầm` (`/compare`) để phân biệt các nhóm 2–3 từ gần nghĩa theo Bài 1–20: trang So sánh (bảng so sánh, ví dụ có nghe đọc, mẹo, tìm kiếm trên mọi bài), trang Luyện tập chọn từ và trang Ôn các nhóm cần ôn; dấu "Đã nắm / Cần ôn" đồng bộ qua tài khoản. Đợt này có nội dung Bài 1–4.

**Architecture:** Logic thuần nằm trong `src/compare.js` (không React, test bằng `node --test`), dùng lại các hàm đã tổng quát hoá của `src/grammar.js` (`filterPoints`/`countPoints` nhận `keyOf`, `buildRoundRobin`). Component chọn từ điền trống `ChoiceBlank` được tách ra để ngữ pháp và so sánh dùng chung.

**Tech Stack:** React 19, Vite 8, `node --test`, router History API tự viết (`src/router.jsx`), Cloudflare Pages + Hono (không đổi).

```js
// The points matching the filter; when none match, the whole lesson (fellBack), like the word lists.
// `keyOf` maps a point id to its marks key (grammar by default; the compare section passes its own).
export function filterPoints(points, filter, getMark, keyOf = grammarKey) {
  const list = points.filter((p) => {
    const m = getMark(keyOf(p.id));
    return filter === 'all' || (filter === 'new' ? m === '' : m === filter);
  });
  return list.length || !points.length ? { points: list, fellBack: false } : { points, fellBack: true };
}

export function countPoints(points, getMark, keyOf = grammarKey) {
  let k = 0;
  let r = 0;
  points.forEach((p) => {
    const m = getMark(keyOf(p.id));
    if (m === 'k') k++;
    else if (m === 'r') r++;
  });
  return { n: points.length, k, r, fresh: points.length - k - r };
}
```

Thay hàm `buildGrammarQuiz` bằng:

```js
// Every question of the given items ({ id, questions }), each passed through prepare(q, id, rng), shuffled.
// With `max`, questions are taken round-robin over the items (in random order) so one item with many
// questions cannot fill the run.
export function buildRoundRobin(items, prepare, { max = Infinity, rng = Math.random } = {}) {
  const queues = shuffle(items, rng).map((it) => shuffle(it.questions, rng).map((q) => prepare(q, it.id, rng)));
  const picked = [];
  for (let round = 0; picked.length < max; round++) {
    const row = queues.filter((qs) => round < qs.length).map((qs) => qs[round]);
    if (!row.length) break;
    picked.push(...row.slice(0, max - picked.length));
  }
  return shuffle(picked, rng);
}

export const buildGrammarQuiz = (points, opts = {}) => buildRoundRobin(points, prepare, opts);
```

```js
const GANG = [['刚', 'gāng'], ['刚才', 'gāngcái']];

// A well-formed group: 2 examples per word, 4 questions alternating the answer.
function pair(id, words = GANG) {
  return {
    id,
    words,
    common: 'C',
    rows: [
      ['Từ loại', ...words.map(() => 'x')],
      ['Vị trí', ...words.map(() => 'y')],
```

```js
// Confusable word groups and their practice logic, with no React so Node tests can use them.
import RAW from './data/compare.json' with { type: 'json' };
import { GRAMMAR_FILTERS, buildRoundRobin, countPoints, filterPoints } from './grammar.js';
import { shuffle } from './random.js';

export { FILTER_NAMES, lessonOf } from './grammar.js';

// lesson ('L1') -> its groups. Lessons not written yet have no key (or []).
export const COMPARE = RAW;
export const ALL_PAIRS = Object.values(COMPARE).flat();
export const PAIR_BY_ID = Object.fromEntries(ALL_PAIRS.map((p) => [p.id, p]));

const NONE = [];
export const pairsOf = (lesson) => COMPARE[lesson] || NONE;

// Group marks live in the vocabulary marks store; the prefix keeps them apart from hanzi and grammar.
export const compareKey = (id) => 'c:' + id;
export const pairTitle = (p) => p.words.map((w) => w[0]).join(' / ');

export const COMPARE_FILTERS = GRAMMAR_FILTERS;
const FILTER_KEY = 'hsk4_compare_filter';
const FILTER_IDS = GRAMMAR_FILTERS.map(([f]) => f);

export function loadCompareFilter() {
  try {
    const f = localStorage.getItem(FILTER_KEY);
    return FILTER_IDS.includes(f) ? f : 'all';
  } catch {
    return 'all';
  }
}

export function saveCompareFilter(f) {
  try {
    localStorage.setItem(FILTER_KEY, f);
  } catch {
    /* private mode / quota: the filter just isn't remembered */
  }
}

export const filterPairs = (pairs, filter, getMark) => filterPoints(pairs, filter, getMark, compareKey);
export const countPairs = (pairs, getMark) => countPoints(pairs, getMark, compareKey);
export const reviewPairs = (getMark) => ALL_PAIRS.filter((p) => getMark(compareKey(p.id)) === 'r');

// For LessonTabs: known groups / groups of each lesson.
export const compareProgress = (getMark) => (lesson) => {
  const c = countPairs(pairsOf(lesson), getMark);
  return { done: c.k, total: c.n };
};

// Pinyin without tone marks, case, spaces or apostrophes ("Suǒ Yǒu" -> "suoyou", "lǜ" -> "lu").
export const toneless = (s) =>
  String(s)
    .normalize('NFD')
    .replace(/[̀-ͯ]/g, '')
    .replace(/[\s'’]/g, '')
    .toLowerCase();

// Groups with a word whose hanzi contains the query, or whose toneless pinyin contains it.
export function searchPairs(query, pairs = ALL_PAIRS) {
  const q = String(query).trim();
  if (!q) return [];
  const plain = toneless(q);
  return pairs.filter((p) => p.words.some(([hz, py]) => hz.includes(q) || toneless(py).includes(plain)));
}

// The choices of a question are always the words of its group.
const withOptions = (p) => ({ id: p.id, questions: p.questions.map((q) => ({ ...q, options: p.words.map((w) => w[0]) })) });
const prepare = (q, pairId, rng) => ({ ...q, pairId, options: shuffle(q.options, rng) });

export const buildCompareQuiz = (pairs, opts = {}) => buildRoundRobin(pairs.map(withOptions), prepare, opts);

// The same questions again (e.g. the wrong ones), in a new order with the options reshuffled.
export const retryCompareQuiz = (questions, rng = Math.random) => shuffle(questions, rng).map((q) => prepare(q, q.pairId, rng));

// ---------- data checks (used by the tests) ----------

const isText = (s) => typeof s === 'string' && s.trim() !== '';

function checkPair(p, lesson, home, order) {
  const e = [];
  const at = p.id;
  const words = Array.isArray(p.words) ? p.words : [];
  const hanzi = words.map((w) => (Array.isArray(w) ? w[0] : undefined));
  if (words.length < 2 || words.length > 3) e.push(`${at}: cần 2–3 từ`);
  if (!words.every((w) => Array.isArray(w) && w.length === 2 && w.every(isText))) e.push(`${at}: mỗi từ phải là [hán tự, pinyin]`);
  if (new Set(hanzi).size !== hanzi.length) e.push(`${at}: từ bị trùng`);
  for (const f of ['common', 'tip']) if (!isText(p[f])) e.push(`${at}: thiếu ${f}`);

  const rows = Array.isArray(p.rows) ? p.rows : [];
  if (rows.length < 2) e.push(`${at}: cần ≥ 2 dòng so sánh`);
  rows.forEach((r, j) => {
    if (!Array.isArray(r) || r.length !== words.length + 1 || !r.every(isText)) e.push(`${at}: dòng ${j + 1} phải có ${words.length + 1} ô`);
  });

  const ex = Array.isArray(p.examples) ? p.examples : [];
  ex.forEach((x, j) => {
    const ok = Array.isArray(x) && x.length === 4 && x.slice(0, 3).every(isText) && Number.isInteger(x[3]) && isText(hanzi[x[3]]);
    if (!ok) e.push(`${at}: ví dụ ${j + 1} phải là [hán tự, pinyin, nghĩa, chỉ số từ]`);
    else if (!x[0].includes(hanzi[x[3]])) e.push(`${at}: ví dụ ${j + 1} không chứa ${hanzi[x[3]]}`);
  });
  hanzi.forEach((h, k) => {
    if (isText(h) && ex.filter((x) => Array.isArray(x) && x[3] === k).length < 2) e.push(`${at}: ${h} cần ≥ 2 ví dụ`);
  });

  const qs = Array.isArray(p.questions) ? p.questions : [];
  if (qs.length < 4 || qs.length > 6) e.push(`${at}: cần 4–6 câu hỏi`);
  qs.forEach((q, j) => {
    const qa = `${at} câu ${j + 1}`;
    if (typeof q.q !== 'string' || q.q.split('___').length !== 2) e.push(`${qa}: q phải có đúng một ___`);
    if (!hanzi.includes(q.answer)) e.push(`${qa}: answer không phải từ của nhóm`);
    if (!isText(q.explain)) e.push(`${qa}: thiếu explain`);
  });
  hanzi.forEach((h) => {
    if (isText(h) && !qs.some((q) => q.answer === h)) e.push(`${at}: ${h} chưa là đáp án câu nào`);
  });

  // A group lives in the lesson of its newest HSK4 word (words outside vocab.json, e.g. HSK1-3, don't count).
  if (!hanzi.some((h) => home[h] === lesson)) e.push(`${at}: không từ nào thuộc từ vựng ${lesson}`);
  const later = hanzi.filter((h) => home[h] && order.indexOf(home[h]) > order.indexOf(lesson));
  if (later.length) e.push(`${at}: ${later.join(', ')} thuộc bài sau, nên chuyển nhóm sang bài đó`);
  return e;
}

// Every problem in the compare data, as readable messages; [] when it is valid. `vocab` is vocab.json.
export function validateCompare(data, vocab) {
  const order = Object.keys(vocab);
  const home = {}; // hanzi -> first lesson whose vocabulary has it
  for (const l of order) for (const w of vocab[l].words) home[w[0]] ??= l;
  const errors = [];
  const seen = new Set();
  for (const [lesson, pairs] of Object.entries(data)) {
    if (!order.includes(lesson)) errors.push(`${lesson}: không có trong vocab.json`);
    if (!Array.isArray(pairs)) {
      errors.push(`${lesson}: phải là mảng`);
      continue;
    }
    if (pairs.length && (pairs.length < 3 || pairs.length > 5)) errors.push(`${lesson}: cần 3–5 nhóm, có ${pairs.length}`);
    pairs.forEach((p, i) => {
      const id = `${lesson.slice(1)}.${i + 1}`;
      if (p.id !== id) errors.push(`${lesson}[${i}]: id phải là "${id}", đang là "${p.id}"`);
      if (seen.has(p.id)) errors.push(`id trùng: ${p.id}`);
      seen.add(p.id);
      errors.push(...checkPair(p, lesson, home, order));
    });
  }
  return errors;
}
```

| 1.2 | 从来 / 一直 | 从来 gần như chỉ đi với phủ định (从来不 / 从来没…过), nói “từ trước đến nay”, không dùng cho tương lai. 一直 dùng cả khẳng định (一直很忙), chỉ liên tục không đổi, kéo được tới tương lai (我会一直等你), còn nghĩa “thẳng” (一直走). |
| 1.3 | 即使 / 虽然 | 即使 là giả định (dù cho… chưa xảy ra), vế sau dùng 也. 虽然 là sự thật đã có, vế sau dùng 但是/可是/却. |
| 1.4 | 熟悉 / 了解 | 熟悉: quen thuộc vì tiếp xúc nhiều (熟悉的声音, 熟悉环境), làm định ngữ “quen thuộc” được, và “làm quen với” (熟悉一下工作). 了解: biết, nắm thông tin/tình hình, còn nghĩa “tìm hiểu” (去了解一下情况). Câu hỏi: 熟悉的人/声音 → 熟悉; 我去___一下情况 → 了解. |
| 2.1 | 却 / 而 / 但是 | 却: phó từ, đứng sau chủ ngữ vế sau, nhấn điều trái dự đoán. 但是: liên từ, đứng đầu vế sau, trước chủ ngữ. 而: liên từ nối hai vế/hai đối tượng đối chiếu (văn viết), không phải “nhưng” trái mong đợi. 但是…却 dùng cùng được. |
| 2.2 | 尽管 / 即使 | 尽管: sự thật, vế sau 但是/可是/还是/却. 即使: giả định, vế sau 也. |
| 2.3 | 理解 / 了解 | 理解: hiểu và thông cảm lý do, cảm xúc, cách làm (我理解你的心情). 了解: biết thông tin, tình hình, kiến thức (了解中国文化, 了解情况); có nghĩa “tìm hiểu”. |
| 2.4 | 专门 / 特别 | 专门: chuyên, riêng cho một mục đích (专门为你做的, 专门研究…), không chỉ mức độ. 特别: rất, đặc biệt (特别好), làm tính từ được (很特别). Câu hỏi: ___好吃 → 特别; ___为他准备 → 专门. |
| 3.1 | 改变 / 变化 | 改变: động từ mang tân ngữ, chủ động làm thay đổi (改变计划, 改变自己). 变化: chủ yếu danh từ (很大的变化); làm động từ thì không mang tân ngữ (天气变化很快). |
| 3.2 | 感觉 / 觉得 | 感觉: danh từ (一种感觉, 感觉不错 = cảm giác) và động từ (cảm thấy qua giác quan). 觉得: chỉ là động từ, thường dùng nêu ý kiến. Câu hỏi chỉ dùng vị trí danh từ (有一种___, 我的___) → 感觉, hoặc nêu ý kiến/đánh giá trong khẩu ngữ khi 感觉 nghe gượng (你___这个办法怎么样？) → 觉得. Không dùng câu kiểu 我___有点儿冷 (cả hai đều đúng). |
| 3.3 | 另外 / 其他 | 另外: “ngoài ra” làm liên từ đầu câu (另外，…), hoặc 另外 + số/lượng từ + danh từ (另外一个人). 其他: “những … khác”, đứng thẳng trước danh từ (其他人, 其他问题) hoặc đứng một mình (还有其他的吗). Câu hỏi: ___，我还想说一件事 → 另外; 除了他，___人都来了 → 其他. |
| 3.4 | 不管 / 尽管 | 不管: theo sau là câu có từ để hỏi, dạng chính phản hoặc lựa chọn (不管多忙 / 不管去不去), vế sau 都/也. 尽管: theo sau là một sự thật cụ thể, vế sau 但是/还是. |
| 4.1 | 以为 / 认为 | 以为: tưởng là (thường sai với thực tế: 我以为…，原来…). 认为: cho rằng, nêu nhận định, trang trọng hơn; dùng được 被认为, 我们认为. Câu hỏi dùng ngữ cảnh 原来/没想到 → 以为; ý kiến chính thức → 认为. |
| 4.2 | 本来 / 原来 | Cả hai: “vốn dĩ, ban đầu”. 原来 có nghĩa “hoá ra” (phát hiện điều trước không biết), 本来 không có. 本来 có nghĩa “lẽ ra / vốn dĩ phải thế” (本来就应该…), 原来 không có. Câu hỏi: ___是你啊！→ 原来; 这件事___就是你的责任 → 本来. Tránh câu chỉ “ban đầu” (cả hai đều đúng). |
| 4.3 | 及时 / 准时 / 按时 | 及时: kịp thời, đúng lúc cần, làm tính từ được (这场雨来得很及时, 及时发现问题). 准时: đúng giờ chính xác, làm vị ngữ được (他很准时). 按时: theo thời gian đã định, chỉ làm trạng ngữ trước động từ (按时吃药, 按时交作业), không làm vị ngữ. Câu hỏi: 他很___ → 准时; 来得很___ / ___发现 → 及时; 医生让他每天___吃药 → 按时. |
| 4.4 | 按照 / 根据 | 按照: làm theo (quy định, kế hoạch, yêu cầu) → hành động tuân theo. 根据: dựa vào (số liệu, tình hình) để kết luận, phán đoán; còn là danh từ “căn cứ” (有什么根据？). Câu hỏi: 你这么说有什么___？→ 根据; ___调查结果，我们发现… → 根据; 请___老师的要求做 → 按照. |

```jsx
const MARKS = [

// The sentence with every occurrence of `word` in bold.
function highlight(zh, word) {
  return zh.split(word).flatMap((s, i) => (i ? [<b key={i}>{word}</b>, s] : [s]));
}

// One group of confusable words. The title row toggles the details: what they share, the comparison
// table (one column per word), examples per word (with speech), the tip and the marks. The element id
// is the group id, for /compare#1.3 links. `lessonLabel` is shown on search results.
export default function ComparePair({ pair, mark, open, onToggleOpen, onMark, speech, lessonLabel }) {
  const cls = 'word-row' + (open ? ' open' : '') + (mark === 'k' ? ' is-known' : mark === 'r' ? ' is-review' : '');
  const speakBtn = (text, id, label) =>
    speech.supported && (
      <button type="button" className={'speak-btn mini' + (speech.speakingId === id ? ' speaking' : '')} aria-label={label} onClick={() => speech.speak(text, id)}>
        🔊
      </button>
    );
  return (
    <div id={pair.id} className={cls}>
      <div className="word-main">
        <button type="button" className="word-toggle" aria-expanded={open} onClick={onToggleOpen}>
          <span className="cp-title">
            {pair.words.map(([hz, py], i) => (
              <span key={hz} className="cp-word">
                {i > 0 && <span className="cp-sep">⇄</span>}
                <span className="cp-hz">{hz}</span>
                <span className="cp-py">{py}</span>
              </span>
            ))}
          </span>
          {lessonLabel && <span className="cp-lesson">{lessonLabel}</span>}
          {mark && <span className={'gp-status ' + mark}>{mark === 'k' ? '✅ Đã nắm' : '🔁 Cần ôn'}</span>}
        </button>
      </div>
      {open && (
        <div className="word-detail">
          <div className="gp-meaning">{pair.common}</div>
          <div className="cp-table-wrap">
            <table className="cp-table">
              <thead>
                <tr>
                  <th scope="col">Tiêu chí</th>
                  {pair.words.map(([hz, py]) => (
                    <th key={hz} scope="col">
                      <span className="cp-hz">{hz}</span> {speakBtn(hz, `cp-${pair.id}-${hz}`, 'Nghe ' + hz)}

                </tr>
              </thead>
              <tbody>

                  <tr key={i}>
```

- [ ] **Step 3: Tạo `src/pages/ComparePage.jsx`**

```jsx
import { useEffect, useMemo, useRef, useState } from 'react';
import ComparePair from '../components/ComparePair.jsx';
import LessonTabs from '../components/LessonTabs.jsx';
import { useApp } from '../context/AppContext.jsx';
import {
  COMPARE_FILTERS,
  PAIR_BY_ID,
  compareKey,
  compareProgress,
  countPairs,
  filterPairs,
  lessonOf,
  loadCompareFilter,
  pairsOf,
  reviewPairs,
  saveCompareFilter,
  searchPairs,
} from '../compare.js';
import { LESSON_COLORS, LESSON_LABELS, VOCAB } from '../lessons.js';
import { Link } from '../router.jsx';

// The group named by the URL hash (/compare#1.3), if it exists.
function hashPair() {
  try {
    const id = decodeURIComponent(window.location.hash.slice(1));
    return PAIR_BY_ID[id] ? id : null;
  } catch {
    return null;
  }
}

// The confusable-word groups of the chosen lesson, filtered by mark, or the search results over every
// lesson. A /compare#<id> link (from a practice run) switches to that group's lesson, shows and opens
// it even if the filter hides it, and scrolls to it.
export default function ComparePage() {
  const { lesson, selectLesson, marks, speech, viewVersion } = useApp();
  const { getMark, toggleMark, markRef } = marks;
  const [filter, setFilter] = useState(loadCompareFilter);
  const [query, setQuery] = useState('');
  const [target, setTarget] = useState(hashPair);
  const [open, setOpen] = useState(() => new Set(target ? [target] : []));
  const scrolled = useRef(false);

  useEffect(() => {
    if (target && lessonOf(target) !== lesson) selectLesson(lessonOf(target));
    // only on mount
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);
  // navigate() has already scrolled to the top, so scroll once the target card is rendered.
  useEffect(() => {
    if (!target || scrolled.current || lessonOf(target) !== lesson) return;
    scrolled.current = true;
    document.getElementById(target)?.scrollIntoView({ block: 'start' });
  }, [target, lesson]);

  const all = pairsOf(lesson);
  // Rebuilt when the lesson, filter or account data changes, not on every mark, so a card does not
  // vanish right after you mark it.
  const { points: pairs, fellBack } = useMemo(() => {
    const m = markRef.current;
    const res = filterPairs(all, filter, (k) => (m[k] && m[k].s) || '');
    if (target && lessonOf(target) === lesson && !res.points.some((p) => p.id === target)) {
      return { points: all.filter((p) => p.id === target || res.points.includes(p)), fellBack: res.fellBack };
    }
    return res;
    // marks are read on purpose only when the view is rebuilt
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [all, filter, target, lesson, viewVersion]);

  const searching = query.trim() !== '';
  const results = useMemo(() => (searching ? searchPairs(query) : []), [searching, query]);

  const counts = countPairs(all, getMark);
  const n = { all: counts.n, new: counts.fresh, r: counts.r, k: counts.k };
  const reviewCount = reviewPairs(getMark).length;

  const pickFilter = (f) => {
    setFilter(f);
    saveCompareFilter(f);
    setTarget(null);
  };
  const pickLesson = (l) => {
    setTarget(null);
    setQuery('');
    selectLesson(l);
  };
  const toggleOpen = (id) =>
    setOpen((s) => {
      const next = new Set(s);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });
  const card = (p, lessonLabel) => (
    <ComparePair
      key={p.id}
      pair={p}
      mark={getMark(compareKey(p.id))}
      open={open.has(p.id)}
      onToggleOpen={() => toggleOpen(p.id)}
      onMark={(s) => toggleMark(compareKey(p.id), s)}
      speech={speech}
      lessonLabel={lessonLabel}
    />
  );

  return (
    <div className="panel">
      <LessonTabs lesson={lesson} getMark={getMark} onSelect={pickLesson} progress={compareProgress(getMark)} />
      <div className="cp-search">
        <input
          type="search"
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          placeholder="Tìm từ trên cả 20 bài: 刚才, gangcai…"
          aria-label="Tìm nhóm từ"
        />
      </div>
      {searching ? (
        <>
          <div className="filter-note">
            {results.length ? `${results.length} nhóm khớp “${query.trim()}” trên cả 20 bài` : 'Không tìm thấy nhóm nào.'}
          </div>
          <div className="grammar-list">{results.map((p) => card(p, LESSON_LABELS[lessonOf(p.id)]))}</div>
        </>
      ) : (
        <>
          <div className="lesson-title" style={{ color: LESSON_COLORS[lesson].a }}>
            {VOCAB[lesson].title}
          </div>
          {all.length > 0 && (
            <div className="filter-bar">
              <div className="filter-chips" role="group" aria-label="Lọc nhóm từ theo trạng thái">
                {COMPARE_FILTERS.map(([f, label]) => (
                  <button key={f} className={'chip' + (f === filter ? ' active' : '')} onClick={() => pickFilter(f)}>
                    {label}
                    <span className="n">{n[f]}</span>
                  </button>
                ))}
              </div>
              {fellBack && <div className="audio-note filter-note">Chưa có nhóm nào ở mục này nên đang hiện cả bài.</div>}
              <div className="filter-tools">
                <span>
                  Đã nắm {counts.k}/{counts.n} nhóm từ của bài này
                </span>
              </div>
            </div>
          )}
          <div className="grammar-actions">
            {all.length > 0 && (
              <Link to="/compare/practice" className="btn">
                📝 Luyện tập bài này
              </Link>
            )}
            {reviewCount > 0 && (
              <Link to="/compare/review" className="btn secondary">
                🔁 Ôn các nhóm cần ôn ({reviewCount})
              </Link>
            )}
          </div>
          {all.length === 0 ? (
            <div className="grammar-empty">Nội dung bài này đang được soạn.</div>
          ) : (
            <div className="grammar-list">{pairs.map((p) => card(p))}</div>
          )}
        </>
      )}
    </div>
  );
}
```

```jsx
// Fill the blank with one of the options; after answering, the right option is green and a wrong pick red.
export default function ChoiceBlank({ q, answered, onAnswer }) {
  const [before, after] = q.q.split('___');
  return (
    <>
      <div className="fb-sentence">
        {before}
        <span className="fb-blank">{answered ? q.answer : '＿＿＿'}</span>
        {after}
      </div>
      <div className="fb-options">
        {q.options.map((opt) => {
          let cls = 'fb-opt';
          if (answered) {
            if (opt === q.answer) cls += ' correct';
            else if (opt === answered.given) cls += ' wrong';
          }
          return (
            <button key={opt} type="button" className={cls} onClick={() => !answered && onAnswer(opt)}>
              {opt}
            </button>
          );
        })}
      </div>
    </>
  );
}
```

```jsx
import { useState } from 'react';
import ChoiceBlank from '../components/ChoiceBlank.jsx';
import LessonTabs from '../components/LessonTabs.jsx';
import { useApp } from '../context/AppContext.jsx';
import {
  FILTER_NAMES,
  PAIR_BY_ID,
  buildCompareQuiz,
  compareKey,
  compareProgress,
  filterPairs,
  loadCompareFilter,
  pairTitle,
  pairsOf,
  retryCompareQuiz,
  reviewPairs,
} from '../compare.js';
import { LESSON_LABELS } from '../lessons.js';
import { Link } from '../router.jsx';

const REVIEW_MAX = 30;
const compareLink = (id) => '/compare#' + id;

// One practice run: every question of the chosen lesson's groups under the compare filter (scope
// "lesson"), or of every "needs review" group, at most 30 questions (scope "review"). Built once on
// mount; the shell remounts the page (key = viewVersion) when the lesson or account data changes.
export default function ComparePracticePage({ scope }) {
  const { lesson, selectLesson, marks } = useApp();
  const { getMark, setMark, toggleMark } = marks;
  const review = scope === 'review';
  const max = review ? REVIEW_MAX : Infinity;

  const [source] = useState(() => {
    const filter = loadCompareFilter();
    const picked = review ? { points: reviewPairs(getMark), fellBack: false } : filterPairs(pairsOf(lesson), filter, getMark);
    return { ...picked, filter };
  });
  const fresh = (questions, run) => ({ questions, run, index: 0, answers: {}, showDone: false });
  const [state, setState] = useState(() => fresh(buildCompareQuiz(source.points, { max }), 0));
  const restart = (questions) => setState((s) => fresh(questions, s.run + 1));
  const go = (i) => setState((s) => ({ ...s, index: i, showDone: false }));
  const finish = () => setState((s) => ({ ...s, showDone: true }));

  const { questions, run, index, answers, showDone } = state;
  const n = questions.length;
  const answerList = Object.values(answers);
  const answeredCount = answerList.length;
  const score = answerList.filter((a) => a.ok).length;

  let body;
  if (!n) {
    body = (
      <div className="fillblank-box quiz-done">
        <div className="filter-note">{review ? 'Chưa có nhóm từ nào cần ôn. 🎉' : 'Nội dung bài này đang được soạn.'}</div>
        <div className="controls">
          <Link to="/compare" className="btn secondary">
            ⚖️ Về So sánh
          </Link>
        </div>
      </div>
    );
  } else if (showDone) {
    const wrongQs = questions.filter((_, i) => answers[i] && !answers[i].ok);
    const wrongIds = [...new Set(wrongQs.map((q) => q.pairId))];
    const runIds = [...new Set(questions.map((q) => q.pairId))];
    body = (
      <div className="fillblank-box quiz-done">
        <h2>
          Kết quả: {score} / {n} (Đã làm {answeredCount} / {n} câu)
        </h2>
        {wrongIds.length ? (
          <div className="quiz-wrong">
            <div className="quiz-label">Nhóm từ có câu sai</div>
            {wrongIds.map((id) => (
              <div key={id} className="quiz-reveal bad">
                <Link to={compareLink(id)}>⚖️ {pairTitle(PAIR_BY_ID[id])}</Link>
              </div>
            ))}
          </div>
        ) : answeredCount > 0 ? (
          <div className="srs-note">🎉 Không sai câu nào!</div>
        ) : (
          <div className="filter-note">Bạn chưa trả lời câu nào.</div>
        )}
        {review && (
          <div className="grammar-review-marks">
            <div className="quiz-label">Đánh dấu “Đã nắm” để bỏ khỏi danh sách cần ôn</div>
            {runIds.map((id) => {
              const key = compareKey(id);
              return (
                <div key={id} className="grammar-review-row">
                  <span>{pairTitle(PAIR_BY_ID[id])}</span>
                  <button type="button" className={'mark-btn' + (getMark(key) === 'k' ? ' active' : '')} data-s="k" onClick={() => toggleMark(key, 'k')}>
                    ✅ Đã nắm
                  </button>
                </div>
              );
            })}
          </div>
        )}
        <div className="controls">
          {wrongQs.length > 0 && (
            <button className="btn" onClick={() => restart(retryCompareQuiz(wrongQs))}>
              🔁 Làm lại câu sai
            </button>
          )}
          <button className="btn" onClick={() => restart(buildCompareQuiz(source.points, { max }))}>
            🔀 Làm lượt mới
          </button>
          <Link to="/compare" className="btn secondary">
            ⚖️ Về So sánh
          </Link>
        </div>
      </div>
    );
  } else {
    const q = questions[index];
    const answered = answers[index] || null;
    const answer = (given) => {
      if (answered) return;
      const ok = given === q.answer;
      const key = compareKey(q.pairId);
      if (!ok && getMark(key) !== 'r') setMark(key, 'r');
      setState((s) => ({ ...s, answers: { ...s.answers, [index]: { given, ok } } }));
    };
    body = (
      <div className="fillblank-box">
        <div className="progress">
          Câu {index + 1} / {n} · Chọn từ đúng điền vào chỗ trống
        </div>
        <ChoiceBlank key={run + ':' + index} q={q} answered={answered} onAnswer={answer} />
        {answered && (
          <div className={'quiz-reveal ' + (answered.ok ? 'ok' : 'bad')}>
            <div>{answered.ok ? '✅ Đúng' : '❌ Sai'}</div>
            <div className="quiz-reveal-sub">{q.explain}</div>
            <div className="quiz-reveal-sub">
              <Link to={compareLink(q.pairId)}>⚖️ Xem so sánh: {pairTitle(PAIR_BY_ID[q.pairId])}</Link>
            </div>
          </div>
        )}
        <div className="controls">
          <button className="btn secondary" onClick={() => go((index - 1 + n) % n)}>
            ◀ Câu trước
          </button>
          {index + 1 < n ? (
            <button key={'next' + index} className="btn" autoFocus={!!answered} onClick={() => go(index + 1)}>
              Câu tiếp ▶
            </button>
          ) : (
            <button key={'done' + index} className="btn" autoFocus={!!answered} onClick={finish}>
              Xem kết quả ▶
            </button>
          )}
          {answeredCount > 0 && index + 1 < n && (
            <button className="btn secondary" onClick={finish}>
              🏁 Kết quả ({score}/{answeredCount})
            </button>
          )}
        </div>
        <div className="fb-score">
          Đã làm: {answeredCount} / {n} · Đúng: {score}
        </div>
      </div>
    );
  }

  return (
    <div className="panel">
      {!review && <LessonTabs lesson={lesson} getMark={getMark} onSelect={selectLesson} progress={compareProgress(getMark)} />}
      <div className="exercise-top">
        <Link to="/compare" className="link-btn">
          ← So sánh
        </Link>
        <span className="exercise-summary">
          {review ? 'Ôn các nhóm cần ôn' : `${LESSON_LABELS[lesson]} · ${FILTER_NAMES[source.filter]}`} · {source.points.length} nhóm · {n} câu
        </span>
      </div>
      {source.fellBack && <div className="filter-note">Chưa có nhóm nào ở mục “{FILTER_NAMES[source.filter]}” nên đang luyện cả bài.</div>}
      {body}
      {n > 0 && (
        <div className="question-grid-wrap">
          <div className="question-grid-header">
            <span className="quiz-label">Danh sách câu</span>
            <span className="question-grid-stat">
              Đã làm {answeredCount}/{n} (Đúng {score})
            </span>
          </div>
          <div className="question-grid">
            {questions.map((_, i) => {
              const a = answers[i];
              const cls = 'q-num' + (i === index && !showDone ? ' current' : '') + (a ? (a.ok ? ' correct' : ' wrong') : '');
              return (
                <button key={i} type="button" className={cls} onClick={() => go(i)} aria-label={`Câu ${i + 1}`}>
                  {i + 1}
                </button>
              );
            })}
          </div>
        </div>
      )}
    </div>
  );
}
```
